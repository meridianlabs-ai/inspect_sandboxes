1"""Separate stdout and stderr for Daytona exec. 

2 

3Daytona's exec API returns one merged output stream, so exec() does not hand 

4the command's streams to the API directly: a wrapper shell runs the command 

5with stdout and stderr piped into two private temp files, then prints both 

6files base64-encoded between per-call sentinels. The command's streams go to 

7the collectors, never to the API stream, so nothing it prints can land in the 

8frame, base64 keeps the sentinels out of the data, and the pipes make the 

9wrapper wait, as the API itself did, until every writer of the command's 

10streams (background children included) has closed them. 

11""" 

12

26# wrapper's housekeeping subshells; the command keeps its own environment.

89def build_capture_command(command: str, capture: ExecCapture) -> str: 

90 """Wrap shell *command* so its stdout and stderr come back separately. 

91 

92 The wrapper creates the three temp files ``0600`` and exclusively (``umask 

93 077`` and ``set -C`` in a subshell, so the command's own umask and options 

94 are untouched; a stale file from a killed attempt is removed first, a 

95 planted entry makes creation fail), opens them on descriptors 3/4/9 for 

96 writing and 6/7/8 for reading, and unlinks them before the command runs: 

97 the command cannot reach them by name, and nothing is left behind if the 

98 process tree is killed. The command runs in a subshell with those 

99 descriptors closed, its stdout and stderr piped into two ``cat`` collectors 

100 that append to the files. Its exit status is written to the status file. 

101 Then ``<stdout>``, base64(stdout), ``<stderr>``, base64(stderr), ``<end>`` 

102 are printed and the wrapper exits with the command's status. A collector 

103 that could not store its stream (a full disk, a file-size limit) prints 

104 ``<failed>`` on the API stream itself, since the scratch filesystem may 

105 have no room for a record, and ``<failed>`` replaces the frame when the 

106 status record is missing or the encoding fails, so a truncated stream or 

107 a lost status is never delivered as the command's result. Its own 

108 programs are resolved through ``SYSTEM_PATH``. Decode the output with :func:`parse_captured_output`. 

109 

110 The pipes only close once every writer has, so a background child's late 

111 output is collected too, and a daemon started without redirecting its 

112 stdout and stderr keeps the exec waiting until the timeout. 

113 """ 

114 q = shlex.quote 

115 out_file, err_file, status_file = ( 

116 q(capture.out_file), 

117 q(capture.err_file), 

118 q(capture.status_file), 

119 ) 

120 start, mid, end = ( 

121 q(capture.stdout_sentinel), 

122 q(capture.stderr_sentinel), 

123 q(capture.end_sentinel), 

124 ) 

125 failed = q(capture.failed_sentinel) 

126 pin = f"PATH={SYSTEM_PATH}; export PATH" 

127 files = f"{out_file} {err_file} {status_file}" 

128 return ( 

129 # setup: private, exclusive files; open them; unlink them 

130 f"if ({pin}; umask 077; set -C; rm -f {files}" 

131 f" && : > {out_file} && : > {err_file} && : > {status_file})" 

132 f" && exec 3>>{out_file} 4>>{err_file} 9>>{status_file}" 

133 f" 6<{out_file} 7<{err_file} 8<{status_file}" 

134 f" && ({pin}; rm -f {files}); then " 

135 # run: stdout -> pipe -> cat -> fd 3 (out); stderr -> pipe -> cat -> fd 4 (err). 

136 # A collector that cannot store its stream (a full disk) prints the 

137 # failure sentinel on its own stderr, the API stream: the scratch 

138 # filesystem may have no room for a record of it. 

139 f"{{ {{ ( ({command}) 3>&- 4>&- 5>&- 6<&- 7<&- 8<&- 9>&-; echo $? >&9; )" 

140 f" 2>&5 | ({pin}; cat >&3 || printf %s {failed} >&2); }} 5>&1" 

141 f" | ({pin}; cat >&4 || printf %s {failed} >&2); }}; " 

142 # a missing or garbled status record (it could not be written) is a 

143 # collection failure too, never the command's result 

144 f"_ec=; read -r _ec <&8; case $_ec in ''|*[!0-9]*) _ec= ;; esac; " 

145 # emit: both streams base64, framed; a collection failure is marked 

146 f'{{ test -n "$_ec" && ({pin}; printf %s {start} && base64 <&6' 

147 f" && printf %s {mid} && base64 <&7 && printf %s {end}); }}" 

148 f" || printf %s {failed}; " 

149 f"exit ${{_ec:-1}}; fi; exit 1" 

150 )

153def parse_captured_output(output: str, capture: ExecCapture) -> tuple[str, str]: 

154 """Split the output of a :func:`build_capture_command` run into (stdout, stderr). 

155 

156 Both streams are base64 in the frame, so a sentinel can never occur inside 

157 them and both come back byte for byte (decoded as UTF-8, invalid bytes 

158 replaced). Text outside the frame is not the command's stdout or stderr 

159 (those go to the collectors) but diagnostics printed by ``sudo`` or the wrapper shell, 

160 such as ``sudo: unable to resolve host``; it is added to stderr, around 

161 the command's own. Whitespace-only text there is the API's and dropped. 

162 

163 Raises: 

164 OutputNotCapturedError: The frame's first sentinel is missing, so the 

165 command never ran. 

166 OutputCollectionError: The command ran, but the wrapper marked the 

167 collection failed, or the frame is incomplete (a truncated 

168 response), duplicated or not valid base64. 

169 """ 

170 # Only the wrapper prints the failure marker; the shell's own error text 

171 # may land before or after it, so look for it anywhere. 

172 if capture.failed_sentinel in output: 

173 raise OutputCollectionError( 

174 "Command ran but its output could not be collected: " 

175 + _without_sentinels(output, capture) 

176 ) 

177 if capture.stdout_sentinel not in output: 

178 raise OutputNotCapturedError( 

179 "Command output was not captured (the exec wrapper did not reach " 

180 f"the command): {output.strip()}" 

181 ) 

182 

183 start = output.find(capture.stdout_sentinel) 

184 mid = output.find(capture.stderr_sentinel) 

185 end = output.find(capture.end_sentinel) 

186 frame = [capture.stdout_sentinel, capture.stderr_sentinel, capture.end_sentinel] 

187 if any(output.count(s) != 1 for s in frame) or not start < mid < end: 

188 raise OutputCollectionError( 

189 "Command ran but its output frame arrived incomplete " 

190 f"({len(output)} characters received)" 

191 ) 

192 

193 try: 

194 stdout = _decode_stream(output[start + len(capture.stdout_sentinel) : mid]) 

195 stderr = _decode_stream(output[mid + len(capture.stderr_sentinel) : end]) 

196 except ValueError as e: 

197 raise OutputCollectionError( 

198 f"Command ran but its output could not be decoded: {e}" 

199 ) from e 

200 before = _diagnostics(output[:start]) 

201 after = _diagnostics(output[end + len(capture.end_sentinel) :]) 

202 return stdout, before + stderr + after

205def captured_exec_result( 

206 exit_code: int, output: str, capture: ExecCapture 

207) -> ExecResult[str]: 

208 """The :class:`ExecResult` of a :func:`build_capture_command` run. 

209 

210 Output without the frame means the wrapper never reached the command (for 

211 example ``sudo: unknown user``, or ``/tmp`` not writable), so the result is 

212 a failed exec with the diagnostics on stderr, as a provider with native 

213 streams would report a failed user switch. The command cannot produce 

214 this itself: it runs with both streams redirected into the collectors. 

215 A collection failure after the command ran raises 

216 :class:`OutputCollectionError`. 

217 """ 

218 try: 

219 stdout, stderr = parse_captured_output(output, capture) 

220 except OutputNotCapturedError: 

221 return ExecResult( 

222 success=False, 

223 returncode=exit_code if exit_code != 0 else 1, 

224 stdout="", 

225 stderr=output, 

226 ) 

227 return ExecResult( 

228 success=exit_code == 0, returncode=exit_code, stdout=stdout, stderr=stderr 

229 )

232def build_remove_command(files: Sequence[str]) -> str: 

233 """A shell command removing temp *files*, with ``rm`` resolved via ``SYSTEM_PATH``.""" 

234 quoted = " ".join(shlex.quote(f) for f in files) 

235 return f"PATH={SYSTEM_PATH}; export PATH; rm -f {quoted}"

238def _decode_stream(encoded: str) -> str: 

239 data = base64.b64decode("".join(encoded.split()), validate=True) 

240 return data.decode("utf-8", errors="replace")

247def _without_sentinels(output: str, capture: ExecCapture) -> str: 

248 for sentinel in capture.sentinels: 

249 output = output.replace(sentinel, " ") 

250 return " ".join(output.split())
